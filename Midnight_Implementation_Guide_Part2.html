<script lang="ts">
import Vue from 'vue';
import midnightStore, { UnshieldedCoin } from '@/stores/midnightStore';
import Messaging from '@/chrome/messaging';
import { MessageTypes } from '@/types/messages';

export default Vue.extend({
  name: 'DustRegistrationDialog',

  props: {
    value: {
      type: Boolean,
      default: false
    }
  },

  data() {
    return {
      loading: false,
      pendingChanges: new Map<string, boolean>(), // coinId -> registered
      pendingOperations: [] as string[]
    };
  },

  computed: {
    show: {
      get(): boolean {
        return this.value;
      },
      set(val: boolean) {
        this.$emit('input', val);
      }
    },

    unshieldedCoins(): UnshieldedCoin[] {
      const nightCoins = midnightStore.coins.unshielded.get('NIGHT') || [];
      return nightCoins.filter(coin => coin.status === 'available');
    },

    dustGenerationRate(): number {
      return midnightStore.dustStats.dustGenerationRate;
    },

    totalRegistered(): bigint {
      return midnightStore.dustStats.totalRegistered;
    },

    totalGenerated(): bigint {
      return midnightStore.dustStats.dustGenerated;
    },

    maxCap(): bigint {
      return this.totalRegistered / 2n; // 50% cap
    },

    hasChanges(): boolean {
      return this.pendingChanges.size > 0;
    }
  },

  methods: {
    formatNight(value: bigint): string {
      const night = Number(value) / 1e12; // 12 decimals (10^12)
      return `${night.toFixed(2)} NIGHT`;
    },

    formatDust(value: bigint): string {
      const dust = Number(value) / 1e12; // 12 decimals (10^12)
      return `${dust.toFixed(2)} DUST`;
    },

    formatTimeSince(timestamp: number): string {
      const now = Date.now();
      const diff = now - timestamp;
      const days = Math.floor(diff / 86400000);
      return days > 0 ? `${days}d ago` : 'today';
    },

    calculateMaxDust(coin: UnshieldedCoin): bigint {
      return coin.value / 2n; // Max DUST = 50% of NIGHT
    },

    calculateDustGenerated(coin: UnshieldedCoin): bigint {
      if (!coin.registeredForDustGeneration) return 0n;

      const elapsed = Date.now() - coin.ctime;
      const days = elapsed / 86400000; // ms to days

      const generatedPerDay = Number(coin.value) * this.dustGenerationRate / 1e12; // 12 decimals (10^12)
      const totalGenerated = BigInt(Math.floor(generatedPerDay * days * 1e12)); // 12 decimals (10^12)

      const maxDust = this.calculateMaxDust(coin);
      return totalGenerated > maxDust ? maxDust : totalGenerated;
    },

    calculateDustProgress(coin: UnshieldedCoin): number {
      const generated = this.calculateDustGenerated(coin);
      const max = this.calculateMaxDust(coin);
      return Number((generated * 100n) / max);
    },

    toggleCoinRegistration(coin: UnshieldedCoin) {
      const newStatus = !coin.registeredForDustGeneration;
      this.pendingChanges.set(coin.id, newStatus);
    },

    async saveChanges() {
      if (!this.hasChanges) return;

      this.loading = true;

      try {
        const changes = Array.from(this.pendingChanges.entries());

        for (const [coinId, shouldRegister] of changes) {
          const coin = this.unshieldedCoins.find(c => c.id === coinId);
          if (!coin) continue;

          this.pendingOperations.push(coinId);

          if (shouldRegister) {
            // Register for DUST generation
            await this.registerCoin(coin);
          } else {
            // Deregister from DUST generation
            await this.deregisterCoin(coin);
          }

          this.pendingOperations = this.pendingOperations.filter(id => id !== coinId);
        }

        this.pendingChanges.clear();

        this.$emit('success', 'DUST registration updated successfully');
        this.close();

      } catch (error) {
        console.error('Registration failed:', error);
        this.$emit('error', error.message || 'Failed to update DUST registration');
      } finally {
        this.loading = false;
      }
    },

    async registerCoin(coin: UnshieldedCoin) {
      const result = await Messaging.sendToBackground({
        method: MessageTypes.MIDNIGHT_REGISTER_FOR_DUST,
        data: {
          coinId: coin.id,
          dustAddress: midnightStore.addresses.dust
        }
      });

      if (!result.success) {
        throw new Error(result.error || 'Registration failed');
      }

      console.log(`✓ Registered coin ${coin.id} for DUST generation`);
    },

    async deregisterCoin(coin: UnshieldedCoin) {
      const result = await Messaging.sendToBackground({
        method: MessageTypes.MIDNIGHT_DEREGISTER_FROM_DUST,
        data: {
          coinId: coin.id
        }
      });

      if (!result.success) {
        throw new Error(result.error || 'Deregistration failed');
      }

      console.log(`✓ Deregistered coin ${coin.id} from DUST generation`);
    },

    async claimGeneratedDust(coin: UnshieldedCoin) {
      // TODO: Implement DUST claim transaction
      console.log('Claim DUST from coin:', coin.id);
    },

    viewCoinDetails(coin: UnshieldedCoin) {
      // TODO: Show coin details dialog
      console.log('Coin details:', coin);
    },

    close() {
      this.pendingChanges.clear();
      this.show = false;
    }
  }
});
</script>

6.3 Midnight Wallet Background Operations

src/chrome/midnightWalletBg.ts (NEW FILE)
import { MidnightWalletService } from '@/services/midnight/midnightWallet.service';
import midnightStore, { midnightActions } from '@/stores/midnightStore';
import { updateCoinRegistration } from '@/db/midnight-db';

class MidnightWalletBg {
  private walletService: MidnightWalletService | null = null;

  /**
   * Initialize Midnight wallet service
   */
  async initialize(walletId: string, network: string, accountIndex: number = 0) {
    this.walletService = new MidnightWalletService(walletId, network, accountIndex);
    console.log('🌙 MidnightWalletBg initialized');
  }

  /**
   * Register NIGHT coin for DUST generation
   */
  async registerForDust(coinId: string, dustAddress: string): Promise<{ txId: string }> {
    if (!this.walletService) {
      throw new Error('Midnight wallet not initialized');
    }

    console.log(`🔐 Registering coin ${coinId} for DUST...`);

    try {
      // 1. Get wallet facade
      const facade = this.walletService.getWalletFacade();

      // 2. Find the NIGHT UTXO
      const coins = midnightStore.coins.unshielded.get('NIGHT') || [];
      const coin = coins.find(c => c.id === coinId);

      if (!coin) {
        throw new Error(`Coin ${coinId} not found`);
      }

      // 3. Build registration transaction
      const utxo = {
        txId: coin.id.split(':')[0],
        index: parseInt(coin.id.split(':')[1]),
        value: coin.value
      };

      // 4. Call SDK registration method
      const txId = await facade.registerNightUtxosForDustGeneration(
        [utxo],
        dustAddress
      );

      console.log(`✓ Registration transaction submitted: ${txId}`);

      // 5. Update database
      await updateCoinRegistration(midnightStore.walletId!, coinId, true);

      // 6. Update store
      const updatedCoins = coins.map(c =>
        c.id === coinId
          ? { ...c, registeredForDustGeneration: true, ctime: Date.now() }
          : c
      );

      midnightActions.updateCoins({
        unshielded: new Map([['NIGHT', updatedCoins]])
      });

      // 7. Add transaction to history
      midnightActions.addTransaction({
        txId,
        type: 'registration',
        status: 'pending',
        timestamp: Date.now(),
        amount: coin.value,
        token: 'NIGHT'
      });

      return { txId };

    } catch (error) {
      console.error('❌ Registration failed:', error);
      throw new Error(`Failed to register for DUST: ${error.message}`);
    }
  }

  /**
   * Deregister NIGHT coin from DUST generation
   */
  async deregisterFromDust(coinId: string): Promise<{ txId: string }> {
    if (!this.walletService) {
      throw new Error('Midnight wallet not initialized');
    }

    console.log(`🔓 Deregistering coin ${coinId} from DUST...`);

    try {
      const facade = this.walletService.getWalletFacade();

      const coins = midnightStore.coins.unshielded.get('NIGHT') || [];
      const coin = coins.find(c => c.id === coinId);

      if (!coin) {
        throw new Error(`Coin ${coinId} not found`);
      }

      const utxo = {
        txId: coin.id.split(':')[0],
        index: parseInt(coin.id.split(':')[1]),
        value: coin.value
      };

      const txId = await facade.deregisterFromDustGeneration([utxo]);

      console.log(`✓ Deregistration transaction submitted: ${txId}`);

      await updateCoinRegistration(midnightStore.walletId!, coinId, false);

      const updatedCoins = coins.map(c =>
        c.id === coinId
          ? { ...c, registeredForDustGeneration: false }
          : c
      );

      midnightActions.updateCoins({
        unshielded: new Map([['NIGHT', updatedCoins]])
      });

      midnightActions.addTransaction({
        txId,
        type: 'deregistration',
        status: 'pending',
        timestamp: Date.now(),
        amount: coin.value,
        token: 'NIGHT'
      });

      return { txId };

    } catch (error) {
      console.error('❌ Deregistration failed:', error);
      throw new Error(`Failed to deregister from DUST: ${error.message}`);
    }
  }

  /**
   * Get wallet service instance
   */
  getWalletService(): MidnightWalletService {
    if (!this.walletService) {
      throw new Error('Midnight wallet not initialized');
    }
    return this.walletService;
  }

  /**
   * Cleanup
   */
  async cleanup() {
    if (this.walletService) {
      await this.walletService.cleanup();
      this.walletService = null;
    }
  }
}

<script lang="ts">
import Vue from 'vue';

export default Vue.extend({
  name: 'ProofProgressIndicator',

  props: {
    value: {
      type: Boolean,
      default: false
    },
    operationId: {
      type: String,
      required: true
    },
    title: {
      type: String,
      default: 'Generating Zero-Knowledge Proof'
    }
  },

  data() {
    return {
      stages: [
        {
          name: 'Preparing',
          description: 'Building transaction structure',
          status: 'preparing'
        },
        {
          name: 'Proving',
          description: 'Generating cryptographic proof (10-15s)',
          status: 'proving'
        },
        {
          name: 'Submitting',
          description: 'Broadcasting to network',
          status: 'submitting'
        }
      ]
    };
  },

  computed: {
    show: {
      get(): boolean {
        return this.value;
      },
      set(val: boolean) {
        this.$emit('input', val);
      }
    },

    provingOperation() {
      return midnightStore.provingOperations.get(this.operationId);
    },

    status(): string {
      return this.provingOperation?.status || 'preparing';
    },

    progress(): number {
      return this.provingOperation?.progress || 0;
    },

    estimatedTimeRemaining(): number | null {
      if (!this.provingOperation?.estimatedCompletion) return null;
      const remaining = Math.max(0, Math.ceil((this.provingOperation.estimatedCompletion - Date.now()) / 1000));
      return remaining;
    },

    stageIcon(): string {
      const icons = {
        'preparing': 'mdi-cog-outline',
        'proving': 'mdi-shield-lock-outline',
        'submitting': 'mdi-send-outline',
        'complete': 'mdi-check-circle',
        'failed': 'mdi-alert-circle'
      };
      return icons[this.status] || 'mdi-help-circle';
    },

    stageColor(): string {
      const colors = {
        'preparing': 'blue',
        'proving': 'purple',
        'submitting': 'indigo',
        'complete': 'success',
        'failed': 'error'
      };
      return colors[this.status] || 'grey';
    },

    stageText(): string {
      const texts = {
        'preparing': 'Preparing Transaction',
        'proving': 'Generating Proof',
        'submitting': 'Submitting to Network',
        'complete': 'Transaction Submitted',
        'failed': 'Operation Failed'
      };
      return texts[this.status] || 'Processing';
    },

    stageDescription(): string {
      const descriptions = {
        'preparing': 'Building transaction with inputs and outputs...',
        'proving': 'Creating zero-knowledge proof (this takes 10-15 seconds)...',
        'submitting': 'Broadcasting transaction to Midnight network...',
        'complete': 'Your transaction has been successfully submitted!',
        'failed': 'An error occurred during proof generation'
      };
      return descriptions[this.status] || '';
    }
  },

  methods: {
    getStageStatus(index: number): 'complete' | 'active' | 'pending' {
      const statusOrder = ['preparing', 'proving', 'submitting', 'complete'];
      const currentIndex = statusOrder.indexOf(this.status);

      if (currentIndex > index) return 'complete';
      if (currentIndex === index) return 'active';
      return 'pending';
    }
  }
});
</script>

<script lang="ts">
import Vue from 'vue';
import midnightStore from '@/stores/midnightStore';
import Messaging from '@/chrome/messaging';
import { MessageTypes } from '@/types/messages';
import ProofProgressIndicator from '../components/ProofProgressIndicator.vue';

export default Vue.extend({
  name: 'SendMidnightDialog',

  components: {
    ProofProgressIndicator
  },

  props: {
    value: {
      type: Boolean,
      default: false
    }
  },

  data() {
    return {
      selectedToken: 'NIGHT',
      transactionType: 'unshielded',
      recipientAddress: '',
      amount: '',
      loading: false,
      showProofProgress: false,
      currentOperationId: '',
      estimatedFee: 0n,

      transactionTypes: [
        {
          value: 'unshielded',
          label: 'Public (Fast)',
          description: 'Visible on blockchain, instant confirmation'
        },
        {
          value: 'shielded',
          label: 'Private (Slow)',
          description: 'Zero-knowledge proof, 10-15 seconds, fully private'
        }
      ],

      rules: {
        required: (v: string) => !!v || 'Required',
        validAddress: (v: string) => {
          if (!v) return true;
          const prefix = this.selectedToken === 'DUST' ? 'midnightt1dust' :
                        this.transactionType === 'shielded' ? 'midnightt1shield' :
                        'midnightt1';
          return v.startsWith(prefix) || `Invalid ${this.selectedToken} address`;
        },
        validAmount: (v: string) => {
          const num = parseFloat(v);
          return (num > 0 && num <= Number(this.availableBalance) / 1e12) || 'Invalid amount'; // 12 decimals (10^12)
        }
      }
    };
  },

  computed: {
    show: {
      get(): boolean {
        return this.value;
      },
      set(val: boolean) {
        this.$emit('input', val);
      }
    },

    availableBalance(): bigint {
      if (this.selectedToken === 'DUST') {
        return midnightStore.balances.dust;
      }
      return this.transactionType === 'shielded'
        ? midnightStore.balances.nightShielded
        : midnightStore.balances.nightUnshielded;
    },

    totalAmount(): bigint {
      const amountBigInt = BigInt(Math.floor(parseFloat(this.amount || '0') * 1e12)); // 12 decimals (10^12)
      return amountBigInt + this.estimatedFee;
    },

    isValid(): boolean {
      return !!(
        this.recipientAddress &&
        this.amount &&
        parseFloat(this.amount) > 0 &&
        this.totalAmount <= this.availableBalance
      );
    }
  },

  watch: {
    amount() {
      this.estimateFee();
    },
    transactionType() {
      this.estimateFee();
    }
  },

  methods: {
    formatBalance(value: bigint): string {
      return (Number(value) / 1e12).toFixed(6); // 12 decimals (10^12)
    },

    formatFee(value: bigint): string {
      return (Number(value) / 1e12).toFixed(6); // 12 decimals (10^12)
    },

    getAddressHint(): string {
      if (this.selectedToken === 'DUST') {
        return 'DUST address (starts with midnightt1dust...)';
      }
      return this.transactionType === 'shielded'
        ? 'Shielded address (starts with midnightt1shield...)'
        : 'Unshielded address (starts with midnightt1...)';
    },

    async pasteAddress() {
      try {
        const text = await navigator.clipboard.readText();
        this.recipientAddress = text;
      } catch (error) {
        console.error('Failed to paste:', error);
      }
    },

    setMaxAmount() {
      const maxWithFee = this.availableBalance - this.estimatedFee;
      this.amount = (Number(maxWithFee) / 1e12).toFixed(6); // 12 decimals (10^12)
    },

    async estimateFee() {
      if (!this.amount) {
        this.estimatedFee = 0n;
        return;
      }

      try {
        const result = await Messaging.sendToBackground({
          method: MessageTypes.MIDNIGHT_ESTIMATE_FEE,
          data: {
            token: this.selectedToken,
            type: this.transactionType,
            amount: BigInt(Math.floor(parseFloat(this.amount) * 1e12)).toString() // 12 decimals (10^12)
          }
        });

        if (result.success) {
          this.estimatedFee = BigInt(result.data.fee);
        }
      } catch (error) {
        console.error('Fee estimation failed:', error);
      }
    },

    async send() {
      if (!this.isValid) return;

      this.loading = true;

      try {
        const amountBigInt = BigInt(Math.floor(parseFloat(this.amount) * 1e12)); // 12 decimals (10^12)

        // Generate operation ID for progress tracking
        this.currentOperationId = `tx-${Date.now()}`;

        // Start proving operation tracking (for shielded only)
        if (this.transactionType === 'shielded') {
          midnightActions.startProvingOperation(this.currentOperationId);
          this.showProofProgress = true;
        }

        const result = await Messaging.sendToBackground({
          method: MessageTypes.MIDNIGHT_SEND_TRANSACTION,
          data: {
            operationId: this.currentOperationId,
            token: this.selectedToken,
            type: this.transactionType,
            to: this.recipientAddress,
            amount: amountBigInt.toString()
          }
        });

        if (result.success) {
          this.$emit('success', `Transaction submitted: ${result.data.txId}`);
          this.close();
        } else {
          throw new Error(result.error);
        }

      } catch (error) {
        console.error('Transaction failed:', error);
        this.$emit('error', error.message || 'Transaction failed');
        this.showProofProgress = false;
      } finally {
        this.loading = false;
      }
    },

    retryTransaction() {
      this.showProofProgress = false;
      this.send();
    },

    cancelTransaction() {
      this.showProofProgress = false;
      this.close();
    },

    close() {
      this.recipientAddress = '';
      this.amount = '';
      this.estimatedFee = 0n;
      this.showProofProgress = false;
      this.show = false;
    }
  }
});
</script>

Appendix B: Troubleshooting Guide

Problem: "Wallets not initialized" Error

Cause: MidnightWalletService.initializeWallets() failed or wasn't called

Solution:

  • Check that encrypted keys are present in database
  • Verify password is correct (decryption succeeds)
  • Check network configuration is valid
  • Look for SDK errors in console (missing dependencies?)

Problem: DUST Registration Failing

Cause: UTXO not available or already registered

Solution:

  • Verify coin.status === 'available'
  • Check coin is NIGHT (not DUST)
  • Ensure coin is unshielded (not in shielded wallet)
  • Check dustAddress is valid
  • Review transaction history for pending registrations

Problem: ZK Proof Generation Stuck

Cause: Proof server unreachable or overloaded

Solution:

  • Check proof server URL is whitelisted in manifest CSP
  • Verify network connectivity (ping proof server)
  • Check proof server status endpoint
  • Increase timeout (may take longer than 15 seconds under load)
  • Implement retry logic with exponential backoff

Problem: Store Not Syncing Across Contexts

Cause: Port connection not established or closed

Solution:

  • Check storeMessaging.subscribe() is called in browser context
  • Verify backgroundStoreMessaging.broadcastUpdate() is called in background
  • Check port is not disconnected (onDisconnect listener)
  • Ensure serialization handles BigInt/Map types correctly
  • Check chrome.storage writes are completing

Appendix C: Performance Optimization

Wallet Login Performance

  • Target: <3 seconds from login to dashboard
  • Optimize key decryption: Run in parallel with Promise.all()
  • Load serialized state: Restore wallets from saved state instead of fresh sync
  • Defer GraphQL connection: Connect in background, don't block login
  • Cache balance data: Show cached balances immediately, update in background

ZK Proof Generation

  • Target: 10-15 seconds (SDK limitation, cannot optimize)
  • User experience: Show detailed progress UI to prevent abandonment
  • Allow minimize: Let users continue browsing while proof generates
  • Batch operations: If sending multiple transactions, batch proofs when possible

DUST Calculation Performance

  • Cache calculations: Don't recalculate DUST accrual on every render
  • Update interval: Recalculate every 60 seconds (DUST generation is slow)
  • Lazy load: Only calculate for visible coins in registration dialog
