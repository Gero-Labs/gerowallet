Phase 1: Three-Key Foundation & Cryptography

📋 Phase Overview

Implement BIP32 hierarchical deterministic key derivation for Midnight's three-key architecture. Each Midnight account requires three separate keys: Zswap (shielded operations), Dust (DUST management), and NightExternal (unshielded NIGHT).

Duration: 2 weeks | Developer: 1 backend engineer

1.1 Understanding Midnight's Three-Key Architecture

Unlike Cardano's single spending key, Midnight requires three derived keys per account:

Key Type Derivation Path Role Constant Purpose
Zswap Key m/44'/2400'/0'/3/0 3 Shielded/privacy operations, ZK proofs
Dust Key m/44'/2400'/0'/2/0 2 DUST registration & management
NightExternal Key m/44'/2400'/0'/0/0 0 Unshielded NIGHT transactions
⚠️ Critical: All Three Keys Required

Unlike Cardano where you can derive addresses from a single key, Midnight wallets are non-functional without all three keys. The DustWallet, ShieldedWallet, and UnshieldedWallet each require their specific key to initialize.

1.2 Implementation: Key Derivation Utility

src/shared/utils/midnightCrypto.ts
import { Bip32PrivateKey } from '@cardano-sdk/crypto';
import { Buffer } from 'buffer';

// BIP44 constants
const HARDENED = 0x80000000;
const PURPOSE = 44;
const MIDNIGHT_COIN_TYPE = 2400;

// Midnight-specific role constants (from Lace analysis)
export enum MidnightRoles {
  NightExternal = 0,  // Unshielded NIGHT transactions
  Dust = 2,           // DUST generation and management
  Zswap = 3           // Shielded operations (zswap circuit)
}

export interface DerivedMidnightKeys {
  zswapKey: Uint8Array;
  dustKey: Uint8Array;
  nightExternalKey: Uint8Array;
  // Store derivation metadata for recovery
  metadata: {
    accountIndex: number;
    coinType: number;
    purpose: number;
  };
}

/**
 * Derives three Midnight keys from a BIP39 seed
 *
 * @param seed - BIP39 entropy (typically from mnemonic)
 * @param accountIndex - Account index (default: 0)
 * @returns Object containing all three derived keys
 *
 * @example
 * const seed = mnemonicToEntropy(mnemonic);
 * const keys = deriveThreeKeys(seed, 0);
 * console.log('Zswap Key:', Buffer.from(keys.zswapKey).toString('hex'));
 */
export function deriveThreeKeys(
  seed: Uint8Array,
  accountIndex: number = 0
): DerivedMidnightKeys {
  console.log('🔑 Deriving Midnight keys for account', accountIndex);

  try {
    // 1. Create root key from BIP39 entropy
    // Empty passphrase (second param) matches Lace's implementation
    const rootKey = Bip32PrivateKey.fromBip39Entropy(
      Buffer.from(seed),
      '' // No passphrase (standard for wallet implementations)
    );

    // 2. Derive purpose level: m/44'
    const purposeKey = rootKey.derive(PURPOSE + HARDENED);
    console.log('  ✓ Purpose key derived (m/44\')');

    // 3. Derive coin type level: m/44'/2400'
    const coinTypeKey = purposeKey.derive(MIDNIGHT_COIN_TYPE + HARDENED);
    console.log('  ✓ Coin type key derived (m/44\'/2400\')');

    // 4. Derive account level: m/44'/2400'/account'
    const accountKey = coinTypeKey.derive(accountIndex + HARDENED);
    console.log(`  ✓ Account key derived (m/44'/2400'/${accountIndex}')`);

    // 5. Derive three role keys
    // Role 0 (NightExternal): m/44'/2400'/account'/0/0
    const nightExternalRoleKey = accountKey.derive(MidnightRoles.NightExternal);
    const nightExternalKey = nightExternalRoleKey.derive(0);
    console.log('  ✓ NightExternal key derived (role 0)');

    // Role 2 (Dust): m/44'/2400'/account'/2/0
    const dustRoleKey = accountKey.derive(MidnightRoles.Dust);
    const dustKey = dustRoleKey.derive(0);
    console.log('  ✓ Dust key derived (role 2)');

    // Role 3 (Zswap): m/44'/2400'/account'/3/0
    const zswapRoleKey = accountKey.derive(MidnightRoles.Zswap);
    const zswapKey = zswapRoleKey.derive(0);
    console.log('  ✓ Zswap key derived (role 3)');

    // 6. Extract raw key bytes (32 bytes each)
    const result: DerivedMidnightKeys = {
      zswapKey: new Uint8Array(zswapKey.rawKey()),
      dustKey: new Uint8Array(dustKey.rawKey()),
      nightExternalKey: new Uint8Array(nightExternalKey.rawKey()),
      metadata: {
        accountIndex,
        coinType: MIDNIGHT_COIN_TYPE,
        purpose: PURPOSE
      }
    };

    console.log('🔑 All three Midnight keys derived successfully');
    return result;

  } catch (error) {
    console.error('❌ Key derivation failed:', error);
    throw new Error(`Failed to derive Midnight keys: ${error.message}`);
  }
}

/**
 * Validates that all three keys are present and correctly sized
 */
export function validateMidnightKeys(keys: DerivedMidnightKeys): boolean {
  const KEY_SIZE = 32; // 256 bits

  if (!keys.zswapKey || keys.zswapKey.length !== KEY_SIZE) {
    console.error('Invalid zswap key size:', keys.zswapKey?.length);
    return false;
  }

  if (!keys.dustKey || keys.dustKey.length !== KEY_SIZE) {
    console.error('Invalid dust key size:', keys.dustKey?.length);
    return false;
  }

  if (!keys.nightExternalKey || keys.nightExternalKey.length !== KEY_SIZE) {
    console.error('Invalid nightExternal key size:', keys.nightExternalKey?.length);
    return false;
  }

  return true;
}

/**
 * Derives public key from private key (for address generation)
 * Used later by wallet SDK
 */
export function derivePublicKey(privateKey: Uint8Array): Uint8Array {
  const bip32Key = Bip32PrivateKey.fromBytes(Buffer.from(privateKey));
  const publicKey = bip32Key.toPublic();
  return new Uint8Array(publicKey.toBytes());
}

1.3 Implementation: EMIP3 Encryption for Keys

💡 Reusing Existing Cardano Infrastructure

Midnight uses the same EMIP3 encryption standard as Cardano. We can leverage Gero's existing encryptWithPassword and decryptWithPassword utilities from src/shared/utils/crypto.ts.

src/shared/utils/midnightCrypto.ts (continued)
import { encryptWithPassword, decryptWithPassword } from '@/shared/utils/crypto';

export interface EncryptedMidnightKeys {
  zswapKey: string;         // Hex-encoded encrypted key
  dustKey: string;          // Hex-encoded encrypted key
  nightExternalKey: string; // Hex-encoded encrypted key
  metadata: {
    accountIndex: number;
    coinType: number;
    purpose: number;
    encryptedAt: number;    // Unix timestamp
  };
}

/**
 * Encrypts all three Midnight keys with EMIP3
 *
 * @param keys - Derived keys from deriveThreeKeys()
 * @param password - User's spending password
 * @returns Encrypted keys as hex strings
 */
export async function encryptMidnightKeys(
  keys: DerivedMidnightKeys,
  password: string
): Promise {
  console.log('🔐 Encrypting Midnight keys with EMIP3...');

  if (!password || password.length === 0) {
    throw new Error('Password cannot be empty');
  }

  if (!validateMidnightKeys(keys)) {
    throw new Error('Invalid keys provided for encryption');
  }

  try {
    // Encrypt each key separately (parallel execution for performance)
    const [encryptedZswap, encryptedDust, encryptedNightExternal] = await Promise.all([
      Promise.resolve(encryptWithPassword(password, keys.zswapKey)),
      Promise.resolve(encryptWithPassword(password, keys.dustKey)),
      Promise.resolve(encryptWithPassword(password, keys.nightExternalKey))
    ]);

    console.log('🔐 All three keys encrypted successfully');

    return {
      zswapKey: encryptedZswap,
      dustKey: encryptedDust,
      nightExternalKey: encryptedNightExternal,
      metadata: {
        ...keys.metadata,
        encryptedAt: Date.now()
      }
    };

  } catch (error) {
    console.error('❌ Key encryption failed:', error);
    throw new Error(`Failed to encrypt Midnight keys: ${error.message}`);
  }
}

/**
 * Decrypts all three Midnight keys with EMIP3
 *
 * @param encryptedKeys - Encrypted keys from storage
 * @param password - User's spending password
 * @returns Decrypted raw key bytes
 */
export async function decryptMidnightKeys(
  encryptedKeys: EncryptedMidnightKeys,
  password: string
): Promise {
  console.log('🔓 Decrypting Midnight keys...');

  if (!password || password.length === 0) {
    throw new Error('Password cannot be empty');
  }

  try {
    // Decrypt each key separately (parallel execution)
    const [zswapKeyBuffer, dustKeyBuffer, nightExternalKeyBuffer] = await Promise.all([
      Promise.resolve(decryptWithPassword(password, encryptedKeys.zswapKey)),
      Promise.resolve(decryptWithPassword(password, encryptedKeys.dustKey)),
      Promise.resolve(decryptWithPassword(password, encryptedKeys.nightExternalKey))
    ]);

    const decryptedKeys: DerivedMidnightKeys = {
      zswapKey: new Uint8Array(zswapKeyBuffer),
      dustKey: new Uint8Array(dustKeyBuffer),
      nightExternalKey: new Uint8Array(nightExternalKeyBuffer),
      metadata: encryptedKeys.metadata
    };

    if (!validateMidnightKeys(decryptedKeys)) {
      throw new Error('Decrypted keys failed validation - incorrect password?');
    }

    console.log('🔓 All three keys decrypted successfully');
    return decryptedKeys;

  } catch (error) {
    console.error('❌ Key decryption failed:', error);
    throw new Error('Failed to decrypt Midnight keys - incorrect password?');
  }
}

1.4 Testing the Key Derivation

src/shared/utils/__tests__/midnightCrypto.test.ts
import { describe, it, expect } from 'vitest';
import { deriveThreeKeys, encryptMidnightKeys, decryptMidnightKeys, validateMidnightKeys } from '../midnightCrypto';
import * as bip39 from 'bip39';

describe('Midnight Key Derivation', () => {
  const TEST_MNEMONIC = 'test walk nut penalty hip pave soap entry language right filter choice';
  const TEST_PASSWORD = 'SecurePassword123!';

  it('should derive three valid keys from seed', () => {
    const seed = bip39.mnemonicToEntropy(TEST_MNEMONIC);
    const keys = deriveThreeKeys(new Uint8Array(Buffer.from(seed, 'hex')), 0);

    expect(keys.zswapKey).toHaveLength(32);
    expect(keys.dustKey).toHaveLength(32);
    expect(keys.nightExternalKey).toHaveLength(32);
    expect(validateMidnightKeys(keys)).toBe(true);
  });

  it('should derive different keys for different accounts', () => {
    const seed = bip39.mnemonicToEntropy(TEST_MNEMONIC);
    const seedBytes = new Uint8Array(Buffer.from(seed, 'hex'));

    const account0 = deriveThreeKeys(seedBytes, 0);
    const account1 = deriveThreeKeys(seedBytes, 1);

    // All keys should be different between accounts
    expect(Buffer.from(account0.zswapKey).toString('hex'))
      .not.toBe(Buffer.from(account1.zswapKey).toString('hex'));
    expect(Buffer.from(account0.dustKey).toString('hex'))
      .not.toBe(Buffer.from(account1.dustKey).toString('hex'));
  });

  it('should encrypt and decrypt keys correctly', async () => {
    const seed = bip39.mnemonicToEntropy(TEST_MNEMONIC);
    const originalKeys = deriveThreeKeys(new Uint8Array(Buffer.from(seed, 'hex')), 0);

    // Encrypt
    const encrypted = await encryptMidnightKeys(originalKeys, TEST_PASSWORD);
    expect(encrypted.zswapKey).toMatch(/^[0-9a-f]+$/); // Hex string
    expect(encrypted.dustKey).toMatch(/^[0-9a-f]+$/);
    expect(encrypted.nightExternalKey).toMatch(/^[0-9a-f]+$/);

    // Decrypt
    const decrypted = await decryptMidnightKeys(encrypted, TEST_PASSWORD);
    expect(Buffer.from(decrypted.zswapKey).toString('hex'))
      .toBe(Buffer.from(originalKeys.zswapKey).toString('hex'));
    expect(Buffer.from(decrypted.dustKey).toString('hex'))
      .toBe(Buffer.from(originalKeys.dustKey).toString('hex'));
    expect(Buffer.from(decrypted.nightExternalKey).toString('hex'))
      .toBe(Buffer.from(originalKeys.nightExternalKey).toString('hex'));
  });

  it('should fail decryption with wrong password', async () => {
    const seed = bip39.mnemonicToEntropy(TEST_MNEMONIC);
    const keys = deriveThreeKeys(new Uint8Array(Buffer.from(seed, 'hex')), 0);
    const encrypted = await encryptMidnightKeys(keys, TEST_PASSWORD);

    await expect(
      decryptMidnightKeys(encrypted, 'WrongPassword')
    ).rejects.toThrow();
  });
});

2.3 Midnight Wallet Service - Core Orchestration Layer

src/services/midnight/midnightWallet.service.ts
import { DustWalletFactory, DustWallet } from '@midnight-ntwrk/wallet-sdk-dust-wallet';
import { ShieldedWalletFactory, ShieldedWallet } from '@midnight-ntwrk/wallet-sdk-shielded';
import { UnshieldedWalletBuilder, UnshieldedWallet } from '@midnight-ntwrk/wallet-sdk-unshielded-wallet';
import { WalletFacade } from '@midnight-ntwrk/wallet-sdk-facade';
import { DustAddress, ShieldedAddress } from '@midnight-ntwrk/wallet-sdk-address-format';
import { deriveThreeKeys, decryptMidnightKeys, EncryptedMidnightKeys } from '@/shared/utils/midnightCrypto';
import { MIDNIGHT_NETWORKS, MidnightNetworkConfig } from '@/utils/networks';
import { Network } from '@/types';

export interface MidnightAddresses {
  dust: string;
  shielded: string;
  unshielded: string;
}

export interface SerializedWalletState {
  dust: string | null;
  shielded: string | null;
  unshielded: string | null;
  unshieldedTxHistory: string | null;
}

/**
 * Main service for managing Midnight wallet lifecycle
 * Orchestrates three separate wallet instances (Dust, Shielded, Unshielded)
 */
export class MidnightWalletService {
  private dustWallet: DustWallet | null = null;
  private shieldedWallet: ShieldedWallet | null = null;
  private unshieldedWallet: UnshieldedWallet | null = null;
  private walletFacade: WalletFacade | null = null;

  private networkConfig: MidnightNetworkConfig;
  private walletId: string;
  private accountIndex: number;

  constructor(walletId: string, network: Network = Network.TESTNET, accountIndex: number = 0) {
    this.walletId = walletId;
    this.accountIndex = accountIndex;
    this.networkConfig = MIDNIGHT_NETWORKS[network];

    console.log(`🌙 MidnightWalletService created for wallet ${walletId}, network: ${network}`);
  }

  /**
   * Initialize wallets from encrypted keys (wallet login flow)
   *
   * @param encryptedKeys - Encrypted keys from database
   * @param password - User's spending password
   * @param serializedState - Optional: Previously saved wallet state for fast recovery
   */
  async initializeFromEncryptedKeys(
    encryptedKeys: EncryptedMidnightKeys,
    password: string,
    serializedState?: SerializedWalletState
  ): Promise {
    console.log('🔓 Initializing Midnight wallets from encrypted keys...');
    const startTime = performance.now();

    try {
      // 1. Decrypt all three keys
      const decryptedKeys = await decryptMidnightKeys(encryptedKeys, password);
      console.log('  ✓ Keys decrypted');

      // 2. Initialize three wallets
      await this.initializeWallets(decryptedKeys, serializedState);

      const elapsed = performance.now() - startTime;
      console.log(`🌙 Midnight wallets initialized in ${elapsed.toFixed(0)}ms`);

      return this.getAddresses();

    } catch (error) {
      console.error('❌ Midnight wallet initialization failed:', error);
      throw new Error(`Failed to initialize Midnight wallets: ${error.message}`);
    }
  }

  /**
   * Initialize wallets from seed (wallet creation flow)
   *
   * @param seed - BIP39 entropy
   * @param password - User's spending password
   * @returns Encrypted keys (to be stored in database)
   */
  async initializeFromSeed(
    seed: Uint8Array,
    password: string
  ): Promise<{ addresses: MidnightAddresses; encryptedKeys: EncryptedMidnightKeys }> {
    console.log('🌱 Initializing Midnight wallets from seed...');
    const startTime = performance.now();

    try {
      // 1. Derive three keys
      const derivedKeys = deriveThreeKeys(seed, this.accountIndex);
      console.log('  ✓ Keys derived');

      // 2. Encrypt keys for storage
      const { encryptMidnightKeys } = await import('@/shared/utils/midnightCrypto');
      const encryptedKeys = await encryptMidnightKeys(derivedKeys, password);
      console.log('  ✓ Keys encrypted');

      // 3. Initialize three wallets
      await this.initializeWallets(derivedKeys);

      const elapsed = performance.now() - startTime;
      console.log(`🌙 Midnight wallets initialized from seed in ${elapsed.toFixed(0)}ms`);

      return {
        addresses: this.getAddresses(),
        encryptedKeys
      };

    } catch (error) {
      console.error('❌ Midnight wallet initialization from seed failed:', error);
      throw new Error(`Failed to initialize Midnight wallets from seed: ${error.message}`);
    }
  }

  /**
   * Internal: Initialize all three wallet instances
   */
  private async initializeWallets(
    keys: { zswapKey: Uint8Array; dustKey: Uint8Array; nightExternalKey: Uint8Array },
    serializedState?: SerializedWalletState
  ): Promise {
    console.log('  Initializing three wallet instances...');

    // Wallet factory configurations
    const dustWalletConfig = {
      networkId: this.networkConfig.networkId,
      indexerClientConnection: {
        indexerHttpUrl: this.networkConfig.indexerHttpUrl,
        indexerWsUrl: this.networkConfig.indexerWsUrl
      },
      provingServerUrl: new URL(this.networkConfig.provingServerUrl),
      relayURL: new URL(this.networkConfig.rpcWsUrl),
      costParameters: this.networkConfig.costParameters
    };

    try {
      // 1. Initialize DustWallet
      if (serializedState?.dust) {
        console.log('    Restoring DustWallet from serialized state...');
        this.dustWallet = DustWalletFactory.restore(serializedState.dust);
      } else {
        console.log('    Creating new DustWallet...');
        this.dustWallet = DustWalletFactory.startWithSeed(
          Buffer.from(keys.dustKey),
          dustWalletConfig
        );
      }
      console.log('    ✓ DustWallet initialized');

      // 2. Initialize ShieldedWallet
      if (serializedState?.shielded) {
        console.log('    Restoring ShieldedWallet from serialized state...');
        this.shieldedWallet = ShieldedWalletFactory.restore(serializedState.shielded);
      } else {
        console.log('    Creating new ShieldedWallet...');
        this.shieldedWallet = ShieldedWalletFactory.startWithShieldedSeed(
          Buffer.from(keys.zswapKey)
        );
      }
      console.log('    ✓ ShieldedWallet initialized');

      // 3. Initialize UnshieldedWallet
      if (serializedState?.unshielded) {
        console.log('    Restoring UnshieldedWallet from serialized state...');
        this.unshieldedWallet = UnshieldedWalletBuilder.restore({
          serializedState: serializedState.unshielded,
          networkId: this.networkConfig.networkId,
          indexerUrl: this.networkConfig.indexerHttpUrl
        });
      } else {
        console.log('    Creating new UnshieldedWallet...');
        const { derivePublicKey } = await import('@/shared/utils/midnightCrypto');
        const publicKey = derivePublicKey(keys.nightExternalKey);

        this.unshieldedWallet = UnshieldedWalletBuilder.build({
          publicKey: Buffer.from(publicKey),
          networkId: this.networkConfig.networkId,
          indexerUrl: this.networkConfig.indexerHttpUrl
        });
      }
      console.log('    ✓ UnshieldedWallet initialized');

      // 4. Create unified WalletFacade
      this.walletFacade = new WalletFacade(
        this.dustWallet,
        this.shieldedWallet,
        this.unshieldedWallet
      );
      console.log('    ✓ WalletFacade created');

    } catch (error) {
      console.error('❌ Wallet initialization failed:', error);
      await this.cleanup();
      throw error;
    }
  }

  /**
   * Get all three addresses
   */
  getAddresses(): MidnightAddresses {
    if (!this.dustWallet || !this.shieldedWallet || !this.unshieldedWallet) {
      throw new Error('Wallets not initialized');
    }

    const dustAddress = DustAddress.codec.encode(
      this.networkConfig.networkId,
      this.dustWallet.capabilities.keys.getDustAddress(this.dustWallet.state)
    );

    const shieldedAddress = ShieldedAddress.codec.encode(
      this.networkConfig.networkId,
      this.shieldedWallet.address
    );

    const unshieldedAddress = this.unshieldedWallet.address;

    return {
      dust: dustAddress.asString(),
      shielded: shieldedAddress.asString(),
      unshielded: unshieldedAddress
    };
  }

  /**
   * Serialize wallet states for persistence
   */
  serializeState(): SerializedWalletState {
    if (!this.dustWallet || !this.shieldedWallet || !this.unshieldedWallet) {
      throw new Error('Wallets not initialized');
    }

    return {
      dust: this.dustWallet.serialize(),
      shielded: this.shieldedWallet.serialize(),
      unshielded: this.unshieldedWallet.serialize(),
      unshieldedTxHistory: null // TODO: Implement tx history serialization
    };
  }

  /**
   * Get wallet facade for transaction operations
   */
  getWalletFacade(): WalletFacade {
    if (!this.walletFacade) {
      throw new Error('Wallet facade not initialized');
    }
    return this.walletFacade;
  }

  /**
   * Cleanup wallet instances
   */
  async cleanup(): Promise {
    console.log('🧹 Cleaning up Midnight wallets...');

    this.dustWallet = null;
    this.shieldedWallet = null;
    this.unshieldedWallet = null;
    this.walletFacade = null;

    console.log('✓ Midnight wallets cleaned up');
  }
}

/**
 * Custom serializer for BigInt and Map types
 */
function serializeStoreUpdates(updates: Partial): any {
  return JSON.parse(JSON.stringify(updates, (key, value) => {
    // Serialize BigInt
    if (typeof value === 'bigint') {
      return value.toString();
    }

    // Serialize Map
    if (value instanceof Map) {
      return {
        __type: 'Map',
        value: Array.from(value.entries())
      };
    }

    return value;
  }));
}

/**
 * Custom deserializer for BigInt and Map types
 */
function deserializeStoreUpdates(serialized: any): any {
  return JSON.parse(JSON.stringify(serialized), (key, value) => {
    // Deserialize BigInt (detect numeric strings for balance fields)
    if (typeof value === 'string' && /^\d+$/.test(value) &&
        (key.includes('balance') || key.includes('Balance') || key.includes('value'))) {
      return BigInt(value);
    }

    // Deserialize Map
    if (value && value.__type === 'Map') {
      return new Map(value.value);
    }

    return value;
  });
}

/**
 * Broadcast updates from background to all browser contexts
 */
function broadcastFromBackground(updates: Partial, immediate = false) {
  if (context === 'background') {
    // Update in-memory state
    Object.assign(midnightStore, updates);

    // Serialize for transmission
    const serialized = serializeStoreUpdates(updates);

    // Broadcast to all connected browser contexts
    backgroundStoreMessaging.broadcastUpdate(STORE_NAME, serialized);

    // Persist to storage (debounced to prevent excessive writes)
    if (immediate) {
      if (storageWriteTimeout) {
        clearTimeout(storageWriteTimeout);
        storageWriteTimeout = null;
      }
      chrome.storage.local.set({ [STORE_NAME]: serializeStoreUpdates(midnightStore) });
      console.log('💾 MidnightStore persisted immediately');
    } else {
      if (storageWriteTimeout) clearTimeout(storageWriteTimeout);
      storageWriteTimeout = setTimeout(() => {
        chrome.storage.local.set({ [STORE_NAME]: serializeStoreUpdates(midnightStore) });
        console.log('💾 MidnightStore persisted (debounced)');
      }, 300);
    }
  }
}

<script lang="ts">
import Vue from 'vue';
import midnightStore, { MidnightTransaction } from '@/stores/midnightStore';
import ReceiveDialog from '../dialogs/ReceiveDialog.vue';
import DustRegistrationDialog from '../dialogs/DustRegistrationDialog.vue';
import SendMidnightDialog from '../dialogs/SendMidnightDialog.vue';

export default Vue.extend({
  name: 'MidnightDashboard',

  components: {
    ReceiveDialog,
    DustRegistrationDialog,
    SendMidnightDialog
  },

  data() {
    return {
      receiveDialog: {
        show: false,
        addressType: 'unshielded' as 'dust' | 'shielded' | 'unshielded',
        address: ''
      },
      dustRegistrationDialog: false,
      sendDialog: false
    };
  },

  computed: {
    balances() {
      return midnightStore.balances;
    },

    dustStats() {
      return midnightStore.dustStats;
    },

    addresses() {
      return midnightStore.addresses;
    },

    networkName(): string {
      const networkMap = {
        'preview': 'Preview Testnet',
        'preprod': 'Preprod Testnet',
        'mainnet': 'Mainnet'
      };
      return networkMap[midnightStore.networkId] || midnightStore.networkId;
    },

    networkColor(): string {
      return midnightStore.networkId === 'mainnet' ? 'success' : 'warning';
    },

    totalValueUSD(): string {
      // TODO: Integrate with price service
      return '$0.00';
    },

    priceChange24h(): string {
      // TODO: Integrate with price service
      return '+0.00%';
    },

    priceChangeClass(): string {
      return 'success--text'; // or 'error--text'
    },

    recentTransactions(): MidnightTransaction[] {
      return midnightStore.transactions.slice(0, 5);
    },

    dustGenerationProgress(): number {
      if (this.dustStats.totalRegistered === 0n) return 0;
      const maxDust = this.dustStats.totalRegistered / 2n; // 50% cap
      return Number((this.dustStats.dustGenerated * 100n) / maxDust);
    }
  },

  methods: {
    formatNight(value: bigint): string {
      const night = Number(value) / 1e12; // CORRECT: 12 decimals (10^12)
      return `${night.toLocaleString(undefined, { maximumFractionDigits: 2 })} NIGHT`;
    },

    formatDust(value: bigint): string {
      const dust = Number(value) / 1e12; // CORRECT: 12 decimals (10^12)
      return `${dust.toLocaleString(undefined, { maximumFractionDigits: 2 })} DUST`;
    },

    formatAmount(amount: bigint, token: string): string {
      return token === 'NIGHT' ? this.formatNight(BigInt(amount)) : this.formatDust(BigInt(amount));
    },

    formatTimestamp(timestamp: number): string {
      const now = Date.now();
      const diff = now - timestamp;

      if (diff < 60000) return 'Just now';
      if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
      if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`;
      return `${Math.floor(diff / 86400000)}d ago`;
    },

    showReceiveDialog(addressType: 'dust' | 'shielded' | 'unshielded') {
      this.receiveDialog.addressType = addressType;
      this.receiveDialog.address = this.addresses[addressType] || '';
      this.receiveDialog.show = true;
    },

    showDustRegistrationDialog() {
      this.dustRegistrationDialog = true;
    },

    showSendDialog() {
      this.sendDialog = true;
    },

    shieldNight() {
      // TODO: Implement shield flow
      console.log('Shield NIGHT');
    },

    showAllTransactions() {
      this.$router.push('/midnight/transactions');
    },

    showTransactionDetails(tx: MidnightTransaction) {
      // TODO: Show transaction details dialog
      console.log('Transaction details:', tx);
    },

    getTransactionIcon(type: string): string {
      const icons = {
        'transfer': 'mdi-swap-horizontal',
        'registration': 'mdi-checkbox-marked-circle',
        'deregistration': 'mdi-close-circle',
        'shield': 'mdi-shield-lock',
        'unshield': 'mdi-shield-off'
      };
      return icons[type] || 'mdi-help-circle';
    },

    getTransactionColor(type: string): string {
      const colors = {
        'transfer': 'primary',
        'registration': 'success',
        'deregistration': 'warning',
        'shield': 'purple',
        'unshield': 'amber'
      };
      return colors[type] || 'grey';
    },

    getTransactionTitle(tx: MidnightTransaction): string {
      const titles = {
        'transfer': `Send ${tx.token}`,
        'registration': 'Registered for DUST',
        'deregistration': 'Unregistered from DUST',
        'shield': 'Shielded NIGHT',
        'unshield': 'Unshielded NIGHT'
      };
      return titles[tx.type] || 'Unknown';
    },

    getStatusColor(status: string): string {
      return status === 'confirmed' ? 'success' : status === 'failed' ? 'error' : 'warning';
    }
  }
});
</script>
